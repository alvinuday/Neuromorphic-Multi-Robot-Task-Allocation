\chapter{QUBO Formulation and Proof}
\label{app:qubo}

\section{Binary to Ising Conversion}

Given a QUBO:
\[
\min_x \quad x^T Q x + p^T x \quad x_i \in \{0,1\}
\]

Substitute $x_k = \frac{1+s_k}{2}$ where $s_k \in \{-1,+1\}$:
\[
x_k x_j = \frac{(1+s_k)(1+s_j)}{4} = \frac{1 + s_k + s_j + s_k s_j}{4}
\]

The Ising objective becomes:
\[
H = \sum_i h_i s_i + \sum_{i<j} J_{ij} s_i s_j
\]

where $h_i = -\frac{w_i}{2}$ and $J_{ij} = \frac{\lambda}{4}$.

\section{Theorem (Penalty Sufficiency)}

\textbf{Theorem 4.1:} If $\lambda > \max_{(i,j) \in E} (w_i + w_j)$, then every global minimum of the QUBO is a feasible MWIS solution.

\textbf{Proof:} Suppose $x^*$ minimizes QUBO but violates constraint $(i,j) \in E$ (i.e., $x_i^* = x_j^* = 1$). Then flipping either variable reduces the objective by at least:
\[
\Delta = \lambda - (w_i + w_j) > 0
\]

Contradiction. Thus all minima are feasible. \hfill $\square$
